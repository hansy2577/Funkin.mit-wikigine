FMS-playAnim

FMS_playAnim() can be used for play custom characters (dad, gf, bf) anim

FMS_playAnim(characterName="string",animName="string",isLooping="boolean");

exemple : 

// play *player* *cheer* animation without looping it
FMS_playAnim("bf","cheer",false);

avansed exemple : 

// play in loop *opponent* *sad* animation every time the player do a good notes hit
function onNotesHit(longNotes) {
  FMS_playAnim("dad","sad",true);
}

-- in jlua
function onNotesHit(longNotes)
  FMS_playAnim("dad","sad",true);
end

FMS-makeElement

FMS_makeElement() can be used for make elements like a boxe, square, circle... for a UI !

FMS_makeElement(id="element",x = 1,y = 1,z = 1,w = 100,h = 100,putIn = "default/GAME/GUI etc..")

exemple : 


    FMS_makeElement("aBoxe",10,100,1,200,200,"GAME")

avansed exemple : 


    let boxe = FMS_makeElement("aBoxe",10,100,1,200,200,"GAME");
    boxe.style.opacity = 0.5;
    boxe.onclick = function () {
      boxe.remove();
    }

FMS-makeText

FMS_makeText() can be used for make a simple Text !

FMS_makeText(id = "element",text = "Hello word !",x = 1,y = 1,z = 1,scale = 1,putIn = "default")

exemple : 


    FMS_makeText("myText","Hello word !",1,1,2,1,"GUI")

avansed exemple : 


    var text = FMS_makeText("myText","Hello word !",1,1,2,1,"GUI");
    var x = 1;
    
    setInterval((/* move the text at the right */) => {
      x++;  // add +1 every loop
      text.style.left = x+"px";
    },10) // 0.1 sec 
  

    local text = FMS_makeText("myText","Hello word !",1,1,2,1,"GUI");
    local x = 1;
    
    loop(function (--[[ move the text at the right ]]) 
      x++; --add +1 every loop
      text.style.left = x+"px";
    end,10) --0.1 sec

FMS_makeText(id = "element",scr = "path/to/my/images",x = 1,y = 1,z = 1,scale = 1,putIn = "default")

exemple : 


    FMS_makeImage("myImages",game.imagesFolder+"/myImages.png",1,1,2,1,"GUI")

    var images = FMS_makeText("myText",game.imagesFolder+"/myImages.png",1,1,2,1,"GUI");

      images.style.visibility = "visible";
      
      setTimeOut((/* wait 1 sec*/) => {
        images.style.visibility = "hidden";
      },1000) // 1 sec
    }
